// Degen level: 18 checks -> a verdict per check (green / blue / red / unknown) and a risk 0..1 -> 6 areas -> 10 points -> level 1..10.
// The colour of a check comes from BAND (exact thresholds on the raw value). The risk number RISK only feeds the score.
function knots(pts) {
  return function (x) {
    if (x <= pts[0][0]) return pts[0][1];
    for (var i = 1; i < pts.length; i++) {
      if (x <= pts[i][0]) { var a = pts[i - 1], b = pts[i]; return a[1] + (b[1] - a[1]) * (x - a[0]) / (b[0] - a[0]); }
    }
    return pts[pts.length - 1][1];
  };
}

  { id: 'earnings',  name: 'Earnings',  ask: 'Is the revenue real?',    w: 2.0 },
  { id: 'liquidity', name: 'Liquidity', ask: 'Can one trade move it?',  w: 1.5 },
  { id: 'heat',      name: 'Heat',      ask: 'Did it already run?',     w: 1.0 },

var has = function (v) { return v !== null && v !== undefined; };

// Verdict per check, read literally from the spec table: "under X" excludes X, "over X" excludes X,
// "X or more" includes X, "A-B" includes both ends. null = cannot be judged.
var BAND = {
  c1: function (i) { if (!has(i.admin)) return null; var a = i.admin; if (a.kind === 'eoa') return 'red'; if (a.kind === 'none') return 'green'; if (a.m <= 1) return 'red'; return (a.m < 3 || a.m / a.n < 0.5) ? 'blue' : 'green'; },
  c2: function (i) { if (!has(i.timelockH)) return null; return i.timelockH < 24 ? 'red' : i.timelockH < 48 ? 'blue' : 'green'; },
  c3: function (i) { if (!has(i.powers)) return null; return i.powers <= 0 ? 'green' : i.powers === 1 ? 'blue' : 'red'; },
  s1: function (i) { if (!has(i.hidden)) return null; return i.hidden < 0.25 ? 'green' : i.hidden <= 1 ? 'blue' : 'red'; },
  s2: function (i) { if (!has(i.emitRate)) return null; return i.emitRate < 0.04 ? 'green' : i.emitRate <= 0.10 ? 'blue' : 'red'; },
  s3: function (i) { if (!has(i.insiderOut)) return null; return i.insiderOut < 0.5 ? 'green' : i.insiderOut <= 1.5 ? 'blue' : 'red'; },
  s4: function (i) { if (!has(i.unlock90)) return null; if (i.cliff) return 'red'; return i.unlock90 < 3 ? 'green' : i.unlock90 <= 8 ? 'blue' : 'red'; },
  e1: function (i) { if (!has(i.realYield)) return null; return i.realYield >= 2 ? 'green' : i.realYield >= 1 ? 'blue' : 'red'; },
  e2: function (i) {
    if (!has(i.phase)) return null;
    if (i.phase === 'declining' || i.phase === 'postcrash' || (has(i.spike) && i.spike > 4.5)) return 'red';
    if (i.phase === 'ignition' || i.phase === 'slowing' || i.phase === 'notrend' || (has(i.spike) && i.spike >= 2.5)) return 'blue';
    return 'green';
  },
  e3: function (i) { if (has(i.holderShare) && i.holderShare < 5) return 'red'; if (!has(i.access)) return null; return i.access === 'direct' ? 'green' : i.access === 'stake' ? 'blue' : 'red'; },
  l1: function (i) { if (!has(i.slip)) return null; return i.slip < 1 ? 'green' : i.slip <= 2.5 ? 'blue' : 'red'; },
  l2: function (i) { if (!has(i.mcapM)) return null; return i.mcapM > 20 ? 'green' : i.mcapM >= 3 ? 'blue' : 'red'; },
  l3: function (i) { if (!has(i.whale)) return null; return i.whale < 2 ? 'green' : i.whale <= 5 ? 'blue' : 'red'; },
  h1: function (i) { if (!has(i.run30)) return null; if (i.run30 > 70 || (i.run30 >= 53 && has(i.athDist) && i.athDist <= 10)) return 'red'; return i.run30 >= 30 ? 'blue' : 'green'; },
  h2: function (i) { if (!has(i.strength)) return null; return i.strength >= 1 ? 'green' : i.strength >= 0.5 ? 'blue' : 'red'; },
  y1: function (i) { if (!has(i.team)) return null; return { doxxed: 'green', anon: 'blue', anonfail: 'red' }[i.team]; },
  y2: function (i) { if (!has(i.audit)) return null; return { deployed: 'green', sister: 'blue', none: 'red' }[i.audit]; },
  y3: function (i) { if (!has(i.track)) return null; return { clean: 'green', repaid: 'blue', dead: 'red' }[i.track]; }

// Risk 0..1 per check, only used inside the score. Its edges line up with the bands (0.25 and 0.6).
var RISK = {
  c1: function (i) { if (!has(i.admin)) return null; var a = i.admin; if (a.kind === 'eoa') return 1; if (a.kind === 'none') return 0; if (a.m <= 1) return 1; return (a.m < 3 || a.m / a.n < 0.5) ? 0.4 : 0.15; },
  c2: function (i) { if (!has(i.timelockH)) return null; var h = i.timelockH; return h <= 0 ? 1 : h < 24 ? 0.6 : h < 48 ? 0.4 : 0.1; },
  c3: function (i) { return has(i.powers) ? Math.min(1, 0.35 * i.powers) : null; },
  s1: function (i) { return has(i.hidden) ? knots([[0, 0], [0.25, 0.25], [1, 0.6], [3, 1]])(i.hidden) : null; },
  s2: function (i) { return has(i.emitRate) ? knots([[0, 0], [0.04, 0.25], [0.10, 0.6], [0.20, 1]])(i.emitRate) : null; },
  s3: function (i) { return has(i.insiderOut) ? knots([[0, 0], [0.5, 0.25], [1.5, 0.6], [3, 1]])(i.insiderOut) : null; },
  s4: function (i) { if (!has(i.unlock90)) return null; var r = knots([[0, 0], [3, 0.25], [8, 0.6], [15, 1]])(i.unlock90); return i.cliff ? Math.max(r, 0.7) : r; },
  e1: function (i) { return has(i.realYield) ? knots([[0.5, 1], [1, 0.6], [2, 0.25], [4, 0]])(i.realYield) : null; },
  e2: function (i) { if (!has(i.phase)) return null; var P = { declining: 1, postcrash: 1, slowing: 0.4, notrend: 0.5, ignition: 0.5, steady: 0.1, accelerating: 0.1 }; var sp = has(i.spike) ? knots([[1.5, 0], [2.5, 0.25], [4.5, 0.6], [8, 1]])(i.spike) : 0; return Math.max(P[i.phase], sp); },
  e3: function (i) { if (has(i.holderShare) && i.holderShare < 5) return 1; if (!has(i.access)) return null; return { direct: 0.1, stake: 0.3, haircut: 0.7, none: 1 }[i.access]; },
  l1: function (i) { return has(i.slip) ? knots([[0.5, 0], [1, 0.25], [2.5, 0.6], [6, 1]])(i.slip) : null; },
  l2: function (i) { return has(i.mcapM) ? knots([[1, 1], [3, 0.6], [10, 0.35], [25, 0.2], [100, 0]])(i.mcapM) : null; },
  l3: function (i) { return has(i.whale) ? knots([[0, 0], [2, 0.25], [5, 0.6], [10, 1]])(i.whale) : null; },
  h1: function (i) { if (!has(i.run30)) return null; var r = knots([[20, 0], [30, 0.25], [70, 0.6], [120, 1]])(i.run30); return Math.min(1, r + (has(i.athDist) && i.athDist <= 10 ? 0.15 : 0)); },
  h2: function (i) { return has(i.strength) ? knots([[0.3, 1], [0.5, 0.6], [1, 0.25], [1.5, 0]])(i.strength) : null; },
  y1: function (i) { return has(i.team) ? { doxxed: 0.1, anon: 0.5, anonfail: 0.8 }[i.team] : null; },
  y2: function (i) { return has(i.audit) ? { deployed: 0.1, sister: 0.5, none: 0.8 }[i.audit] : null; },
  y3: function (i) { return has(i.track) ? { clean: 0.1, repaid: 0.4, dead: 0.7 }[i.track] : null; }

var PILLAR_OF = { c1: 'control', c2: 'control', c3: 'control', s1: 'supply', s2: 'supply', s3: 'supply', s4: 'supply', e1: 'earnings', e2: 'earnings', e3: 'earnings', l1: 'liquidity', l2: 'liquidity', l3: 'liquidity', h1: 'heat', h2: 'heat', y1: 'history', y2: 'history', y3: 'history' };
var REPORT_CHECKS = ['e3', 'y1', 'y2', 'y3'];
// Data quality: too little history or stale data make the time-based checks "not checked" (and they widen the range).
var NEEDS_HISTORY = ['e2', 'h1', 'h2'];                       // apps with under 180 days of data
var GOES_STALE = ['e1', 'e2', 'l1', 'l2', 'h1', 'h2'];        // latest data point over 10 days old
function verdict(r) { return r === null ? 'unknown' : r >= 0.6 ? 'red' : r >= 0.25 ? 'blue' : 'green'; }   // areas only
function pillarF(rs, fill) { // 60% worst + 40% average; a check that is not checked counts as `fill`
  if (!rs.length) return fill;
  var v = rs.map(function (r) { return r === null ? fill : r; });
  var max = Math.max.apply(null, v), mean = v.reduce(function (a, b) { return a + b; }, 0) / v.length;
  return 0.6 * max + 0.4 * mean;
}
function score(input) {
  var why = {}, r = {}, bands = {};
  Object.keys(RISK).forEach(function (k) { r[k] = RISK[k](input); bands[k] = BAND[k](input); });
  if (has(input.historyDays) && input.historyDays < 180) NEEDS_HISTORY.forEach(function (k) { r[k] = null; bands[k] = null; why[k] = 'history'; });
  if (has(input.dataLagDays) && input.dataLagDays > 10) GOES_STALE.forEach(function (k) { r[k] = null; bands[k] = null; why[k] = 'stale'; });
  var na = input.na || [];   // checks that do not apply to this app (for example a timelock on a chain without one)
  na.forEach(function (k) { r[k] = null; bands[k] = 'na'; why[k] = 'na'; });
  Object.keys(r).forEach(function (k) { if (r[k] === null && !why[k]) why[k] = REPORT_CHECKS.indexOf(k) > -1 ? 'report' : 'scan'; });
  function total(fill) {
    return PILLARS.map(function (p) {
      var rs = Object.keys(PILLAR_OF).filter(function (k) { return PILLAR_OF[k] === p.id && na.indexOf(k) < 0; }).map(function (k) { return r[k]; });
      var f = pillarF(rs, fill);
      return { id: p.id, f: f, pts: f * p.w };
    });
  }
  var mid = total(0.5), lo = total(0), hi = total(1);
  var sum = function (a) { return a.reduce(function (s, p) { return s + p.pts; }, 0); };
  // Floors: two facts from the contract scan (not checks of their own; they show in the Admin powers tooltip).
  var floored = !!(input.unverifiedSource || input.eoaMint);
  var floor = floored ? 8 : 1;
  var lvl = function (s) { return Math.min(10, Math.max(floor, Math.round(s))); };
  var unknown = Object.keys(r).filter(function (k) { return r[k] === null && na.indexOf(k) < 0; });
  return { r: r, bands: bands, why: why, pillars: mid, points: sum(mid), level: lvl(sum(mid)), low: lvl(sum(lo)), high: lvl(sum(hi)), unknown: unknown, floored: floored };
}
